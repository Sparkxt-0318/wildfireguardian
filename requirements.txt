# the result against this file and pyproject.toml. These four were imported at

# something else (affine <- rasterio, fonttools <- matplotlib) or as a
# conda-only package named in docs/ENVIRONMENT.md but never in a dependency
# file (h5netcdf, and h5py beneath it). A transitive is not a declaration: the
# intermediate package is free to drop it in any release.

#

# working environment (Linux aarch64, PyPI wheels, Python 3.11) — NOT read off
